// schema.prisma

generator client {
    provider = "prisma-client-js"
}

datasource db {
    provider = "postgresql"
    url      = env("DATABASE_URL")
}

/**
 * =========================
 * Enums
 * =========================
 */

enum Role {
    USER
    LAWYER
    ADMIN
}

enum VerificationType {
    IDENTITY
    BAR_LICENSE
    EMAIL
    PHONE
    EDUCATION
}

enum VerificationStatus {
    PENDING
    VERIFIED
    REJECTED
    EXPIRED
}

enum ConnectionStatus {
    PENDING
    ACCEPTED
    DECLINED
    BLOCKED
}

enum CaseStatus {
    OPEN
    UNDER_REVIEW
    ASSIGNED
    IN_PROGRESS
    COMPLETED
    CANCELLED
}

enum Urgency {
    LOW
    MEDIUM
    HIGH
    CRITICAL
}

enum ApplicationStatus {
    SUBMITTED
    UNDER_REVIEW
    ACCEPTED
    REJECTED
    WITHDRAWN
}

enum MessageRole {
    USER
    ASSISTANT
    SYSTEM
}

enum Visibility {
    PUBLIC
    UNLISTED
    PRIVATE
}

enum PostStatus {
    DRAFT
    PUBLISHED
    ARCHIVED
}

/**
 * =========================
 * Core Accounts
 * =========================
 */

model User {
    id        String   @id @default(cuid())
    email     String   @unique
    name      String?
    handle    String?  @unique
    imageUrl  String?
    role      Role     @default(USER)
    createdAt DateTime @default(now())
    updatedAt DateTime @updatedAt

    // Profile and expertise
    profile        LawyerProfile?
    verifications  Verification[]
    certifications Certification[]
    educations     Education[]
    experiences    WorkExperience[]
    skills         UserSkill[]
    practiceAreas  UserPracticeArea[]

    // Networking
    connectionsInitiated Connection[] @relation("connections_initiated")
    connectionsReceived  Connection[] @relation("connections_received")

    // Cases
    postedCases  Case[]            @relation("PostedCases")
    applications CaseApplication[]
    assignments  CaseAssignment[]

    // Content
    blogPosts BlogPost[]
    comments  Comment[]
    likes     PostLike[]

    // AI / Documents
    chatSessions ChatSession[]
    documents    Document[]
}

model LawyerProfile {
    id                    String   @id @default(cuid())
    userId                String   @unique
    title                 String?
    bio                   String?
    location              String?
    barNumber             String?
    barCouncil            String?
    yearsOfExperience     Int?     @default(0)
    caseCount             Int?     @default(0)
    hourlyRate            Int?
    isAvailableForProBono Boolean  @default(true)
    website               String?
    linkedinUrl           String?
    createdAt             DateTime @default(now())
    updatedAt             DateTime @updatedAt

    user          User               @relation(fields: [userId], references: [id])
    // Redundant via user: certifications/skills/practiceAreas
    practiceAreas UserPracticeArea[]

    @@index([yearsOfExperience])
    @@index([isAvailableForProBono])
}

/**
 * =========================
 * Expertise: Practice Areas & Skills
 * =========================
 */

model PracticeArea {
    id   Int    @id @default(autoincrement())
    name String @unique
    slug String @unique

    users UserPracticeArea[]
    cases CasePracticeArea[]
}

model UserPracticeArea {
    userId         String
    practiceAreaId Int
    level          Int? // 1-5 self-assessed or verified

    user         User         @relation(fields: [userId], references: [id])
    practiceArea PracticeArea @relation(fields: [practiceAreaId], references: [id])

    @@id([userId, practiceAreaId])
}

model Skill {
    id   Int    @id @default(autoincrement())
    name String @unique

    users UserSkill[]
    cases CaseRequiredSkill[]
}

model UserSkill {
    userId  String
    skillId Int
    level   Int? // 1-5

    user  User  @relation(fields: [userId], references: [id])
    skill Skill @relation(fields: [skillId], references: [id])

    @@id([userId, skillId])
}

/**
 * =========================
 * Credentials & Background
 * =========================
 */

model Certification {
    id                 String             @id @default(cuid())
    userId             String
    name               String
    issuer             String
    issuedAt           DateTime?
    expiresAt          DateTime?
    credentialId       String?
    credentialUrl      String?
    verificationStatus VerificationStatus @default(PENDING)
    createdAt          DateTime           @default(now())

    user User @relation(fields: [userId], references: [id])

    @@index([userId])
    @@index([verificationStatus])
}

model Verification {
    id          String             @id @default(cuid())
    userId      String
    type        VerificationType
    status      VerificationStatus @default(PENDING)
    verifiedAt  DateTime?
    notes       String?
    evidenceUrl String?
    createdAt   DateTime           @default(now())

    user User @relation(fields: [userId], references: [id])

    @@unique([userId, type])
    @@index([status])
}

model Education {
    id           String    @id @default(cuid())
    userId       String
    school       String
    degree       String?
    fieldOfStudy String?
    startDate    DateTime?
    endDate      DateTime?
    description  String?

    user User @relation(fields: [userId], references: [id])

    @@index([userId])
}

model WorkExperience {
    id           String    @id @default(cuid())
    userId       String
    organization String
    title        String
    startDate    DateTime?
    endDate      DateTime?
    isCurrent    Boolean   @default(false)
    description  String?

    user User @relation(fields: [userId], references: [id])

    @@index([userId])
    @@index([isCurrent])
}

/**
 * =========================
 * Professional Networking
 * =========================
 */

model Connection {
    id          String           @id @default(cuid())
    requesterId String
    addresseeId String
    status      ConnectionStatus @default(PENDING)
    createdAt   DateTime         @default(now())

    requester User @relation("connections_initiated", fields: [requesterId], references: [id])
    addressee User @relation("connections_received", fields: [addresseeId], references: [id])

    @@unique([requesterId, addresseeId])
    @@index([status])
}

/**
 * =========================
 * Pro Bono Cases
 * =========================
 */

model Case {
    id             String     @id @default(cuid())
    title          String
    description    String
    postedById     String
    status         CaseStatus @default(OPEN)
    urgency        Urgency    @default(MEDIUM)
    location       String?
    remoteAllowed  Boolean    @default(true)
    estimatedHours Int?
    createdAt      DateTime   @default(now())
    updatedAt      DateTime   @updatedAt

    postedBy       User                @relation("PostedCases", fields: [postedById], references: [id])
    applications   CaseApplication[]
    assignments    CaseAssignment[]
    practiceAreas  CasePracticeArea[]
    requiredSkills CaseRequiredSkill[]

    @@index([status])
    @@index([urgency])
    @@index([createdAt])
}

model CaseApplication {
    id          String            @id @default(cuid())
    caseId      String
    applicantId String
    coverLetter String?
    status      ApplicationStatus @default(SUBMITTED)
    createdAt   DateTime          @default(now())
    updatedAt   DateTime          @updatedAt

    applicant User @relation(fields: [applicantId], references: [id])
    case      Case @relation(fields: [caseId], references: [id])

    @@unique([caseId, applicantId])
    @@index([status])
}

model CaseAssignment {
    id          String    @id @default(cuid())
    caseId      String
    lawyerId    String
    assignedAt  DateTime  @default(now())
    startedAt   DateTime?
    completedAt DateTime?
    outcome     String?

    case   Case @relation(fields: [caseId], references: [id])
    lawyer User @relation(fields: [lawyerId], references: [id])

    @@unique([caseId, lawyerId])
}

model CasePracticeArea {
    caseId         String
    practiceAreaId Int

    case         Case         @relation(fields: [caseId], references: [id])
    practiceArea PracticeArea @relation(fields: [practiceAreaId], references: [id])

    @@id([caseId, practiceAreaId])
}

model CaseRequiredSkill {
    caseId  String
    skillId Int
    weight  Int? // importance weighting

    case  Case  @relation(fields: [caseId], references: [id])
    skill Skill @relation(fields: [skillId], references: [id])

    @@id([caseId, skillId])
}

/**
 * =========================
 * Blog / Content Sharing
 * =========================
 */

model BlogPost {
    id            String     @id @default(cuid())
    authorId      String
    title         String
    slug          String     @unique
    excerpt       String?
    content       String
    coverImageUrl String?
    status        PostStatus @default(DRAFT)
    visibility    Visibility @default(PUBLIC)
    publishedAt   DateTime?
    createdAt     DateTime   @default(now())
    updatedAt     DateTime   @updatedAt

    author     User           @relation(fields: [authorId], references: [id])
    comments   Comment[]
    likes      PostLike[]
    categories PostCategory[]
    tags       PostTag[]

    @@index([status])
    @@index([publishedAt])
}

model BlogCategory {
    id    Int            @id @default(autoincrement())
    name  String         @unique
    slug  String         @unique
    posts PostCategory[]
}

model BlogTag {
    id    Int       @id @default(autoincrement())
    name  String    @unique
    slug  String    @unique
    posts PostTag[]
}

model PostCategory {
    postId     String
    categoryId Int

    post     BlogPost     @relation(fields: [postId], references: [id])
    category BlogCategory @relation(fields: [categoryId], references: [id])

    @@id([postId, categoryId])
}

model PostTag {
    postId String
    tagId  Int

    post BlogPost @relation(fields: [postId], references: [id])
    tag  BlogTag  @relation(fields: [tagId], references: [id])

    @@id([postId, tagId])
}

model Comment {
    id        String   @id @default(cuid())
    postId    String
    authorId  String
    content   String
    createdAt DateTime @default(now())
    updatedAt DateTime @updatedAt

    post   BlogPost @relation(fields: [postId], references: [id])
    author User     @relation(fields: [authorId], references: [id])

    @@index([postId])
    @@index([authorId])
}

model PostLike {
    userId    String
    postId    String
    createdAt DateTime @default(now())

    user User     @relation(fields: [userId], references: [id])
    post BlogPost @relation(fields: [postId], references: [id])

    @@id([userId, postId])
}

/**
 * =========================
 * AI Chat: Documents & Sessions
 * =========================
 */

model Document {
    id         String   @id @default(cuid())
    ownerId    String
    title      String?
    url        String
    mimeType   String?
    sizeBytes  Int?
    uploadedAt DateTime @default(now())

    owner    User                  @relation(fields: [ownerId], references: [id])
    sessions ChatSessionDocument[]

    @@index([ownerId])
}

model ChatSession {
    id        String   @id @default(cuid())
    userId    String
    title     String?
    createdAt DateTime @default(now())
    updatedAt DateTime @updatedAt

    user      User                  @relation(fields: [userId], references: [id])
    messages  ChatMessage[]
    documents ChatSessionDocument[]

    @@index([userId])
    @@index([createdAt])
}

model ChatSessionDocument {
    sessionId  String
    documentId String
    addedAt    DateTime @default(now())

    session  ChatSession @relation(fields: [sessionId], references: [id])
    document Document    @relation(fields: [documentId], references: [id])

    @@id([sessionId, documentId])
}

model ChatMessage {
    id        String      @id @default(cuid())
    sessionId String
    role      MessageRole
    content   String
    tokens    Int?
    createdAt DateTime    @default(now())

    session ChatSession @relation(fields: [sessionId], references: [id])

    @@index([sessionId])
}
